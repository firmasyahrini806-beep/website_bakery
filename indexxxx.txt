    <script>
        // 1. Mobile Menu Toggle
        const mobileMenuBtn = document.getElementById('mobile-menu-btn');
        const mobileMenu = document.getElementById('mobile-menu');
        const iconOpen = document.getElementById('icon-open');
        const iconClose = document.getElementById('icon-close');
        const mobileLinks = document.querySelectorAll('.mobile-link');

        function toggleMobileMenu() {
            const isHidden = mobileMenu.classList.contains('hidden');
            if (isHidden) {
                mobileMenu.classList.remove('hidden');
                iconOpen.classList.add('hidden');
                iconClose.classList.remove('hidden');
            } else {
                mobileMenu.classList.add('hidden');
                iconOpen.classList.remove('hidden');
                iconClose.classList.add('hidden');
            }
        }

        mobileMenuBtn.addEventListener('click', toggleMobileMenu);

        // Tutup menu mobile otomatis saat link navigasi diklik
        mobileLinks.forEach(link => {
            link.addEventListener('click', () => {
                mobileMenu.classList.add('hidden');
                iconOpen.classList.remove('hidden');
                iconClose.classList.add('hidden');
            });
        });

        // 2. Filter Kategori Menu
        const filterButtons = document.querySelectorAll('.filter-btn');
        const productCards = document.querySelectorAll('.product-card');

        filterButtons.forEach(btn => {
            btn.addEventListener('click', () => {
                // Reset style tombol aktif
                filterButtons.forEach(b => {
                    b.classList.remove('bg-caramel-500', 'text-cream-50', 'shadow-soft');
                    b.classList.add('bg-cream-100', 'text-cocoa-800');
                });
                
                // Aktifkan tombol yang diklik
                btn.classList.remove('bg-cream-100', 'text-cocoa-800');
                btn.classList.add('bg-caramel-500', 'text-cream-50', 'shadow-soft');

                const filterValue = btn.getAttribute('data-filter');

                productCards.forEach(card => {
                    const cardCategory = card.getAttribute('data-category');
                    if (filterValue === 'all' || cardCategory === filterValue) {
                        card.style.display = 'flex';
                    } else {
                        card.style.display = 'none';
                    }
                });
            });
        });

        // 3. Modal Pemesanan Produk ke WhatsApp
        let selectedProduct = '';
        let selectedPrice = 0;
        let currentQty = 1;

        const orderModal = document.getElementById('order-modal');
        const modalProductName = document.getElementById('modal-product-name');
        const modalProductPrice = document.getElementById('modal-product-price');
        const modalQty = document.getElementById('modal-qty');
        const modalTotal = document.getElementById('modal-total');
        const modalNotes = document.getElementById('modal-notes');

        function formatRupiah(angka) {
            return 'Rp ' + angka.toLocaleString('id-ID');
        }

        function openOrderModal(productName, price) {
            selectedProduct = productName;
            selectedPrice = price;
            currentQty = 1;
            modalNotes.value = '';

            modalProductName.textContent = productName;
            modalProductPrice.textContent = formatRupiah(price) + ' / item';
            modalQty.textContent = currentQty;
            modalTotal.textContent = formatRupiah(selectedPrice * currentQty);

            orderModal.classList.remove('hidden');
            orderModal.classList.add('flex');
        }

        function closeOrderModal() {
            orderModal.classList.add('hidden');
            orderModal.classList.remove('flex');
        }

        function updateQty(change) {
            if (currentQty + change >= 1) {
                currentQty += change;
                modalQty.textContent = currentQty;
                modalTotal.textContent = formatRupiah(selectedPrice * currentQty);
            }
        }

        function sendWhatsAppOrder() {
            const phoneNumber = '6281234567890';
            const notes = modalNotes.value.trim();
            const totalFormatted = formatRupiah(selectedPrice * currentQty);

            let message = `Halo Sweeat Pircuy! Saya ingin memesan:\n\n`;
            message += `• Produk: *${selectedProduct}*\n`;
            message += `• Jumlah: *${currentQty}*\n`;
            message += `• Estimasi Total: *${totalFormatted}*\n`;
            if (notes) {
                message += `• Catatan: ${notes}\n`;
            }
            message += `\nMohon info ketersediaan dan detail pembayarannya ya. Terima kasih!`;

            const whatsappUrl = `https://wa.me/${phoneNumber}?text=${encodeURIComponent(message)}`;
            window.open(whatsappUrl, '_blank', 'noopener,noreferrer');
            closeOrderModal();
        }

        // Tutup modal jika area luar modal diklik
        orderModal.addEventListener('click', (e) => {
            if (e.target === orderModal) {
                closeOrderModal();
            }
        });
    </script>
